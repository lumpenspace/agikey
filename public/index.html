How to Connect Any OpenAI Tool

Agiary works as a drop-in replacement for OpenAI endpoints

Python (openai package)

from openai import OpenAI

client = OpenAI(
    base_url="http://127.0.0.1:8000/v1",
    api_key="agiary"  # any dummy key
)

# 1. Non-streaming Chat Completion
response = client.chat.completions.create(
    model="agy",  # or "claude", "grok", "codex", "gemini-3.8-flash-high"
    messages=[{"role": "user", "content": "Write a python fibonacci function"}],
    stream=False
)
print(response.choices[0].message.content)

# 2. Real-time Streaming
stream = client.chat.completions.create(
    model="agy",
    messages=[{"role": "user", "content": "Explain quicksort"}],
    stream=True
)
for chunk in stream:
    if chunk.choices[0].delta.content:
        print(chunk.choices[0].delta.content, end="", flush=True)

Node.js / JavaScript

import OpenAI from 'openai';

const openai = new OpenAI({
  baseURL: 'http://127.0.0.1:8000/v1',
  apiKey: 'agiary',
});

// Streaming chat completion
const stream = await openai.chat.completions.create({
  model: 'agy',
  messages: [{ role: 'user', content: 'Say hello!' }],
  stream: true,
});

for await (const chunk of stream) {
  process.stdout.write(chunk.choices[0]?.delta?.content || '');
}

cURL (Streaming & Sync)

# Streaming SSE
curl -N http://127.0.0.1:8000/v1/chat/completions \
  -H "Content-Type: application/json" \
  -d '{
    "model": "agy",
    "messages": [{"role": "user", "content": "Hello!"}],
    "stream": true
  }'

# Text / Legacy Completions API
curl http://127.0.0.1:8000/v1/completions \
  -H "Content-Type: application/json" \
  -d '{
    "model": "agy",
    "prompt": "Complete this sentence: In theory, there is no difference between theory and",
    "stream": false
  }'

Conversations API & Sessions

# 1. Continue multi-turn thread via OpenAI API
curl -N http://127.0.0.1:8000/v1/chat/completions \
  -H "Content-Type: application/json" \
  -d '{
    "model": "agy",
    "conversation_id": "conv-12345678",
    "messages": [{"role": "user", "content": "What did we just discuss?"}],
    "stream": true
  }'

# 2. Inspect threads via Agikey CLI
agikey convs
agikey chat -c conv-12345678 "Continue debugging"
